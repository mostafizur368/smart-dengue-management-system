rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return request.auth != null
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "admin";
    }

    function isDoctor() {
      return request.auth != null
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "doctor";
    }

    // Task 12: public users may submit a question for review; they cannot publish it.
    match /question_suggestions/{suggestionId} {
      allow create: if request.resource.data.keys().hasOnly([
          'question', 'normalizedQuestion', 'language', 'status',
          'submittedBy', 'createdAt'
        ])
        && request.resource.data.keys().hasAll([
          'question', 'normalizedQuestion', 'language', 'status',
          'submittedBy', 'createdAt'
        ])
        && request.resource.data.question is string
        && request.resource.data.question.size() >= 8
        && request.resource.data.question.size() <= 300
        && request.resource.data.normalizedQuestion is string
        && request.resource.data.normalizedQuestion.size() >= 1
        && request.resource.data.normalizedQuestion.size() <= 300
        && request.resource.data.language in ['bn', 'en']
        && request.resource.data.status == 'pending'
        && request.resource.data.createdAt is timestamp
        && (
          request.resource.data.submittedBy == null
          || (
            request.auth != null
            && request.resource.data.submittedBy == request.auth.uid
          )
        );
      allow read, update, delete: if isAdmin();
    }

    // Only admins can publish reviewed Q&A. Public users can read approved entries.
    match /community_question_bank/{questionId} {
      allow read: if resource.data.status == 'approved' || isAdmin();
      allow create, update, delete: if isAdmin();
    }

    // Task 11: users can submit explicit chatbot ratings; only admins can review them.
    match /chatbot_feedback/{feedbackId} {
      allow create: if request.resource.data.keys().hasOnly([
          'question', 'answerPreview', 'matchedQuestion', 'rating', 'intent', 'matchScore',
          'matchLevel', 'language', 'userId', 'createdAt', 'createdAtClient',
          'feedbackReason', 'feedbackComment'
        ])
        && request.resource.data.keys().hasAll([
          'question', 'answerPreview', 'rating', 'intent', 'matchScore',
          'matchLevel', 'language', 'userId', 'createdAt', 'createdAtClient'
        ])
        && request.resource.data.question is string
        && request.resource.data.question.size() <= 500
        && request.resource.data.answerPreview is string
        && request.resource.data.answerPreview.size() <= 400
        && (
          !request.resource.data.keys().hasAny(['matchedQuestion'])
          || (
            request.resource.data.matchedQuestion is string
            && request.resource.data.matchedQuestion.size() <= 300
          )
        )
        && request.resource.data.rating in ['helpful', 'not_helpful']
        && request.resource.data.intent is string
        && request.resource.data.intent.size() <= 80
        && request.resource.data.matchScore is number
        && request.resource.data.matchScore >= 0
        && request.resource.data.matchScore <= 100
        && request.resource.data.matchLevel in ['high', 'medium', 'low']
        && request.resource.data.language in ['bn', 'en']
        && (!request.resource.data.keys().hasAny(['feedbackReason']) || request.resource.data.feedbackReason in ['', 'inaccurate', 'irrelevant', 'unclear', 'outdated', 'safety_concern', 'other'])

        && request.resource.data.createdAt is timestamp
        && request.resource.data.createdAtClient is string
        && request.resource.data.createdAtClient.size() <= 40
        && (
          request.resource.data.userId == null
          || (
            request.auth != null
            && request.resource.data.userId == request.auth.uid
          )
        );
      allow read: if isAdmin() || (request.auth != null && resource.data.userId == request.auth.uid);
      allow update: if isAdmin();
      allow delete: if isAdmin() || (request.auth != null && resource.data.userId == request.auth.uid);
    }

      }

      allow create: if isAdmin()
        && request.resource.data.sourceEntryId is string
        && request.resource.data.sourceEntryId.size() <= 180
        && request.resource.data.sourceQuestion is string
        && request.resource.data.sourceQuestion.size() <= 500
        && request.resource.data.answerBn is string
        && request.resource.data.answerBn.size() <= 6000

        && request.resource.data.assignedReviewerUid is string
        && request.resource.data.status == 'pending'

        && request.resource.data.sourceIds.size() <= 10
        && request.resource.data.createdAt is timestamp;
      allow read: if isAdmin() || resource.data.status == 'approved' || isAssignedClinicalReviewer();

        isAssignedClinicalReviewer()
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'status', 'reviewDecision', 'reviewerNotes', 'reviewedSourceIds',
          'reviewerUid', 'reviewerName', 'reviewedAt', 'reviewedAtClient', 'reviewDueAt'

        && request.resource.data.assignedReviewerUid == resource.data.assignedReviewerUid
        && request.resource.data.reviewerUid == request.auth.uid
        && request.resource.data.status in ['approved', 'needs_revision', 'rejected']
        && request.resource.data.reviewDecision == request.resource.data.status
        && request.resource.data.reviewerNotes is string
        && request.resource.data.reviewerNotes.size() >= 10
        && request.resource.data.reviewerNotes.size() <= 2000
        && request.resource.data.reviewedSourceIds is list
        && request.resource.data.reviewedSourceIds.size() <= 10
        && (request.resource.data.status != 'approved' || request.resource.data.reviewedSourceIds.size() >= 1)
        && request.resource.data.reviewedAt is timestamp
      );
      allow delete: if isAdmin();
    }
