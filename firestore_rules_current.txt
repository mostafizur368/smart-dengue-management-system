rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function isAdmin() {
      return request.auth != null
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == "admin";
    }

    match /contact_messages/{document} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /symptom_checks/{document} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /appointments/{appointmentId} {
      function isOwner() {
        return request.auth != null
          && resource.data.userId == request.auth.uid;
      }
      function isAssignedDoctor() {
        return request.auth != null
          && resource.data.doctorUid == request.auth.uid;
      }
      allow read: if isAdmin() || isOwner() || isAssignedDoctor();
      allow create: if request.auth != null
        && request.resource.data.userId == request.auth.uid;
      allow update: if isAdmin()
        || (isAssignedDoctor()
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['status']));
      allow delete: if isAdmin();
    }

    match /users/{userId} {
      allow create: if request.auth != null
        && request.auth.uid == userId
        && request.resource.data.role == "user";
      allow read: if request.auth != null
        && (request.auth.uid == userId || isAdmin());
      allow update, delete: if isAdmin();
    }

    match /doctor_applications/{applicationId} {
      allow create, read, update, delete: if isAdmin();
    }

    match /public_doctors/{doctorId} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /ticket_counts/{countId} {
      allow read: if true;
      allow create, update: if request.auth != null
        && request.resource.data.booked is int
        && request.resource.data.booked >= 0
        && request.resource.data.booked <= 30
        && request.resource.data.total == 30
        && request.resource.data.doctorId is string
        && request.resource.data.appointmentDate is string;
      allow delete: if isAdmin();
    }
  }
}
